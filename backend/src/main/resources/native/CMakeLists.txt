cmake_minimum_required(VERSION 3.24)
project(MarginTraceJni LANGUAGES CXX)
option(MARGIN_CUDA "Build the CUDA JNI bridge" OFF)
option(MARGIN_OPENCL "Build the OpenCL JNI bridge" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(JNI REQUIRED COMPONENTS JVM)
# Hide engine symbols so the two independently loaded runtimes cannot interpose one another.
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_CUDA_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_TESTING OFF CACHE BOOL "Native regressions are built separately" FORCE)
if(MARGIN_CUDA)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    set(FGW_ONLY ON CACHE BOOL "Only FGW targets from the CUDA module" FORCE)
    add_subdirectory(../cuda cuda)
    add_library(margintrace_cuda SHARED acceleration_jni.cpp ../cuda/bom_upward_engine.cu)
    target_compile_definitions(margintrace_cuda PRIVATE MARGIN_CUDA)
    target_include_directories(margintrace_cuda PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(margintrace_cuda PRIVATE cuda_fgw CUDA::cudart)
    set_target_properties(margintrace_cuda PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
    install(TARGETS margintrace_cuda LIBRARY DESTINATION . RUNTIME DESTINATION .)
endif()
if(MARGIN_OPENCL)
    # Exceptions must be enabled consistently in the wrapper and engine.
    add_compile_definitions(CL_HPP_ENABLE_EXCEPTIONS)
    add_subdirectory(../opencl opencl)
    add_library(margintrace_opencl SHARED acceleration_jni.cpp)
    target_include_directories(margintrace_opencl PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(margintrace_opencl PRIVATE bom_upward_engine fgw_opencl_engine)
    install(TARGETS margintrace_opencl LIBRARY DESTINATION . RUNTIME DESTINATION .)
    install(FILES ../opencl/bom_upward.cl ../opencl/fgw.cl DESTINATION .)
endif()
